# Pinned vLLM video-test stack for Laskin (issue #32).
#
# WHY THIS IS NOT JUST `pip install vllm`
# ---------------------------------------
# Laskin's GPUs are Tesla V100-PCIE-32GB = compute capability 7.0 (Volta, sm_70).
# Current vLLM requires compute capability >= 7.5 and its prebuilt wheels are
# built on CUDA >= 12.8, whose arch list starts at sm_75. Installing current vLLM
# here does not run.
#
# The last era whose wheels still carry sm_70 kernels is vLLM 0.8.x. That release
# predates Qwen3-VL, so on Volta the newest usable vision-language family is
# Qwen2.5-VL. This file therefore pins the combination proven to work on the
# V100; see docs/LASKIN_VLLM_VIDEO_TEST.md for the measured evidence.
#
# Install into an ISOLATED venv (scripts/laskin/install_vllm_video_test.sh) so the
# machine's existing CUDA/PyTorch environment is never modified.

vllm==0.28.0
# vLLM 0.8.5 pulls the torch build it was tested against (2.6.0+cu124), which
# still lists sm_70. Do not "upgrade" torch: 2.11+/cu128+ drops Volta.
transformers==5.10.1
tokenizers>=0.21,<0.22
huggingface-hub>=0.30,<1.0

# Video path: process_vision_info + a decoder.
qwen-vl-utils==0.0.14
av==17.1.0

# Support deps that vLLM/transformers need in a clean venv.
pymongo==4.18.2
numpy==2.2.6
pillow==12.3.0
pandas>=2.2,<3
